9/// A `CBOR` data item (RFC 8949).
10///
11/// # Example
12/// ```rust
13/// use cbor_next::DataItem;
14///
15/// let item = DataItem::from(vec![("name", "cbor")]);
16/// let encoded = item.encode();
17/// assert_eq!(DataItem::decode(&encoded).unwrap(), item);
18/// ```
19#[derive(Clone)]
20#[non_exhaustive]
21pub enum DataItem {
22    /// Unsigned integer, major type 0.
23    Unsigned(u64),
24    /// Negative integer, major type 1, stored as `n` for the value `-1 - n`.
25    ///
26    /// Build it with `DataItem::from(-10_i64)`; an `i128` past this range
27    /// becomes a bignum.
28    Signed(u64),
29    /// Byte string, major type 2.
30    Byte(ByteString),
31    /// Text string, major type 3.
32    Text(TextString),
33    /// Array, major type 4.
34    Array(Array),
35    /// Map, major type 5.
36    Map(Map),
37    /// Tag, major type 6.
38    Tag(Tag),
39    /// `false` or `true`, simple values 20 and 21.
40    Boolean(bool),
41    /// `null`, simple value 22.
42    Null,
43    /// `undefined`, simple value 23.
44    Undefined,
45    /// Floating point number of any width, held as `f64`.
46    Floating(f64),
47    /// Any other simple value.
48    GenericSimple(Simple),
49}
50
51/// Bits of an `f64` which hold its significand
52const NAN_SIGNIFICAND: u64 = (1 << 52) - 1;
53
54impl PartialEq for DataItem {
55    fn eq(&self, other: &Self) -> bool {
56        match (self, other) {
57            (Self::Unsigned(first), Self::Unsigned(second))
58            | (Self::Signed(first), Self::Signed(second)) => first == second,
59            (Self::Byte(first), Self::Byte(second)) => first == second,
60            (Self::Text(first), Self::Text(second)) => first == second,
61            (Self::Array(first), Self::Array(second)) => first == second,
62            (Self::Map(first), Self::Map(second)) => first == second,
63            (Self::Tag(first), Self::Tag(second)) => first == second,
64            (Self::Boolean(first), Self::Boolean(second)) => first == second,
65            (Self::Null, Self::Null) | (Self::Undefined, Self::Undefined) => true,
66            (Self::Floating(first), Self::Floating(second)) => {
67                // RFC 8949 section 5.6.1: `-0.0 == 0.0`, NaNs equal by
68                // significand
69                if first.is_nan() || second.is_nan() {
70                    first.is_nan()
71                        && second.is_nan()
72                        && first.to_bits() & NAN_SIGNIFICAND == second.to_bits() & NAN_SIGNIFICAND
73                } else {
74                    first == second
75                }
76            }
77            (Self::GenericSimple(first), Self::GenericSimple(second)) => first == second,
78            _ => false,
79        }
80    }
81}

83impl Eq for DataItem {}

85impl Hash for DataItem {
86    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
87        std::mem::discriminant(self).hash(state);
88        match self {
89            Self::Unsigned(val) | Self::Signed(val) => val.hash(state),
90            Self::Byte(items) => items.hash(state),
91            Self::Text(text) => text.hash(state),
92            Self::Array(values) => values.hash(state),
93            Self::Map(map_content) => {
94                map_content.entries().len().hash(state);
95                // equality ignores entry order, so the hash must too
96                let mut combined: u64 = 0;
97                for entry in map_content.entries() {
98                    let mut entry_hasher = DefaultHasher::new();
99                    entry.hash(&mut entry_hasher);
100                    combined = combined.wrapping_add(entry_hasher.finish());
101                }
102                state.write_u64(combined);
103            }
104            Self::Tag(tag_content) => tag_content.hash(state),
105            Self::Boolean(val) => val.hash(state),
106            Self::Floating(val) => {
107                if val.is_nan() {
108                    (val.to_bits() & NAN_SIGNIFICAND).hash(state);
109                } else {
110                    // `-0.0` and `0.0` hash alike
111                    let normalized = if *val == 0.0 { 0.0 } else { *val };
112                    normalized.to_bits().hash(state);
113                }
114            }
115            Self::GenericSimple(simple_number) => simple_number.hash(state),
116            Self::Null | Self::Undefined => {}
117        }
118    }
119}
