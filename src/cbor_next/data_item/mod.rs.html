9/// Enum representing different types of data item that can be encoded or
10/// decoded in `CBOR` (Concise Binary Object Representation).
11///
12/// `CBOR` is a data format designed for small code and message size, often used
13/// in constrained environments. This `DataItem` enum covers all major types
14/// defined in the `CBOR` specification (RFC 8949).
15///
16/// # Example
17/// ```rust
18/// use cbor_next::DataItem;
19///
20/// let item = DataItem::from(vec![("name", "cbor")]);
21/// let encoded = item.encode();
22/// assert_eq!(DataItem::decode(&encoded).unwrap(), item);
23/// ```
24#[derive(PartialEq, Clone)]
25#[non_exhaustive]
26pub enum DataItem {
27    /// Unsigned integer represented by `CBOR` major type 0.
28    ///
29    /// This variant can hold non-negative integer values up to `u64::MAX`.
30    Unsigned(u64),
31    /// Negative integer represented by `CBOR` major type 1.
32    ///
33    /// This variant stores the absolute value minus one of the negative
34    /// integer. For example, a `CBOR` negative integer representing -1 would
35    /// store `0`, -10 would store `9`. The actual negative value is derived
36    /// as `-(1 + value)`. Building this variant from the value it stands for
37    /// is done with `DataItem::from(-10_i64)` or, for the full `CBOR` range,
38    /// `DataItem::try_from(-10_i128)`.
39    Signed(u64),
40    /// Byte string represented by `CBOR` major type 2.
41    ///
42    /// Contains an arbitrary sequence of bytes.
43    Byte(ByteString),
44    /// UTF-8 string (text string) represented by `CBOR` major type 3.
45    ///
46    /// Contains a sequence of Unicode characters encoded as UTF-8.
47    Text(TextString),
48    /// Array of `CBOR` data items represented by `CBOR` major type 4.
49    ///
50    /// An ordered sequence of zero or more `CBOR` data items.
51    Array(Array),
52    /// Map of `CBOR` key-value pairs represented by `CBOR` major type 5.
53    ///
54    /// Keys within a map must be unique
55    Map(Map),
56    /// Tagged item (semantic tag) represented by `CBOR` major type 6.
57    ///
58    /// Consists of an unsigned integer (the tag) and a single `CBOR` data item
59    /// (the tagged content). Tags provide semantic information about the
60    /// enclosed data item, allowing for type extension
61    /// or application-specific interpretations.
62    Tag(Tag),
63    /// Boolean represented as a simple value within `CBOR` major type 7.
64    ///
65    /// Can be either `true` or `false`.
66    Boolean(bool),
67    /// Null represented as a simple value within `CBOR` major type 7.
68    ///
69    /// Represents the absence of a value.
70    Null,
71    /// Undefined represented as a simple value within `CBOR` major type
72    /// 7.
73    ///
74    /// Distinct from `Null`, it represents an undefined state.
75    Undefined,
76    /// Floating-point number represented as a simple value within `CBOR` major
77    /// type 7.
78    ///
79    /// Can represent half-precision (16-bit), single-precision (32-bit), or
80    /// double-precision (64-bit) floating-point numbers. but locally saves
81    /// data as f64
82    Floating(f64),
83    /// An generic simple value represented by `CBOR` major type 7.
84    ///
85    /// This variant handles simple values that are not explicitly covered by
86    /// `Boolean`, `Null`, `Undefined`, or `Floating`. These generic simple
87    /// values have a numerical representation as defined in the `CBOR`
88    /// specification.
89    GenericSimple(Simple),
90}

94impl Hash for DataItem {
95    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
96        std::mem::discriminant(self).hash(state);
97        match self {
98            Self::Unsigned(val) | Self::Signed(val) => val.hash(state),
99            Self::Byte(items) => items.hash(state),
100            Self::Text(text) => text.hash(state),
101            Self::Array(values) => values.hash(state),
102            Self::Map(map_content) => {
103                map_content.is_indefinite().hash(state);
104                map_content.entries().len().hash(state);
105                // entries are compared without regard to their order, so they
106                // must be hashed with an order independent
107                // combination as well
108                let mut combined: u64 = 0;
109                for entry in map_content.entries() {
110                    let mut entry_hasher = DefaultHasher::new();
111                    entry.hash(&mut entry_hasher);
112                    combined = combined.wrapping_add(entry_hasher.finish());
113                }
114                state.write_u64(combined);
115            }
116            Self::Tag(tag_content) => tag_content.hash(state),
117            Self::Boolean(val) => val.hash(state),
118            Self::Floating(val) => {
119                // normalize -0.0 and 0.0 to the same representation for hashing
120                let normalized = if *val == 0.0 { 0.0 } else { *val };
121                normalized.to_be_bytes().hash(state);
122            }
123            Self::GenericSimple(simple_number) => simple_number.hash(state),
124            Self::Null | Self::Undefined => {}
125        }
126    }
127}
