1//! Numbers of the tags registered by RFC 8949 and RFC 8742.
2//!
3//! A tag number carries no meaning to the encoder or the decoder, which keep
4//! any tag as it is. These names only save looking the numbers up, and the
5//! content each one requires is given with its name.
6//!
7//! # Example
8//! ```rust
9//! use cbor_next::{DataItem, tags};
10//!
11//! let time = DataItem::tag(tags::EPOCH_TIME, 1_363_896_240);
12//! assert_eq!(time.encode(), [0xc1, 0x1a, 0x51, 0x4b, 0x67, 0xb0]);
13//! assert_eq!(time.tag_numbers(), [tags::EPOCH_TIME]);
14//! ```
