1//! Streaming: `CBOR` as tokens, one per head (RFC 8949 section 5.1).
2//!
3//! A [`StreamDecoder`] is fed bytes in any pieces and yields a [`Token`] for
4//! each head it can complete, so nested items and RFC 8742 sequences are seen
5//! as they arrive. A [`StreamEncoder`] writes tokens back to bytes.
6//!
7//! Both check well-formedness, including where each break goes, and the
8//! decoder checks UTF-8. Of the [`Decoder`] settings, the stream applies the
9//! nesting depth, item count and indefinite lengths. It cannot see repeated
10//! map keys, tag content or the deterministic form (section 5.6), so those
11//! are left to the application.
12//!
13//! # Example
14//! ```rust
15//! use cbor_next::{Decoder, Token};
16//!
17//! // [1, "a"] arriving in two pieces
18//! let mut stream = Decoder::new().stream();
19//! stream.feed(&[0x82, 0x01, 0x61]);
20//! assert_eq!(
21//!     stream.next_token().unwrap(),
22//!     Some(Token::ArrayStart(Some(2)))
23//! );
24//! assert_eq!(stream.next_token().unwrap(), Some(Token::Unsigned(1)));
25//! // the text is not complete yet
26//! assert_eq!(stream.next_token().unwrap(), None);
27//! stream.feed(&[0x61]);
28//! assert_eq!(stream.next_token().unwrap(), Some(Token::Text("a".into())));
29//! assert!(stream.is_at_boundary());
30//! ```

38/// Byte which closes an indefinite length item.
39const BREAK: u8 = 0xFF;
40
41/// One head of `CBOR`, with the payload of a definite length string.
42#[derive(Debug, Clone, PartialEq)]
43#[non_exhaustive]
44pub enum Token {
45    /// Unsigned integer, major type 0.
46    Unsigned(u64),
47    /// Negative integer `-1 - n`, major type 1.
48    Signed(u64),
49    /// A definite length byte string, or a chunk of an indefinite one.
50    Bytes(Vec<u8>),
51    /// A definite length text string, or a chunk of an indefinite one.
52    Text(String),
53    /// Start of an indefinite length byte string; chunks follow, then a break.
54    BytesStart,
55    /// Start of an indefinite length text string; chunks follow, then a break.
56    TextStart,
57    /// Start of an array of `n` items, or of indefinite length.
58    ArrayStart(Option<u64>),
59    /// Start of a map of `n` pairs, or of indefinite length.
60    MapStart(Option<u64>),
61    /// A tag number; the tagged item follows.
62    Tag(u64),
63    /// `false` or `true`.
64    Boolean(bool),
65    /// `null`.
66    Null,
67    /// `undefined`.
68    Undefined,
69    /// A float of any width.
70    Floating(f64),
71    /// Any other simple value.
72    Simple(Simple),
73    /// End of an indefinite length item.
74    Break,
75}
76
77/// An open item, waiting for what completes it.
78#[derive(Debug, Clone, Copy)]
79enum Frame {
80    /// Items left, or `None` until a break
81    Array(Option<u64>),
82    /// Pairs left, or `None` until a break, and whether a value is due
83    Map {
84        remaining: Option<u64>,
85        value_next: bool,
86    },
87    /// The tagged item is due
88    Tag,
89    /// Chunks of this major type until a break
90    Chunks(u8),
91}
92
93/// The open items, checked token by token.
94#[derive(Debug, Clone)]
95struct Nesting {
96    frames: Vec<Frame>,
97    limit: usize,
98}
99
100impl Nesting {
101    fn new(limit: usize) -> Self {
102        Self {
103            frames: vec![],
104            limit,
105        }
106    }
107
108    /// The major type of the indefinite string open innermost.
109    fn chunk_type(&self) -> Option<u8> {
110        match self.frames.last() {
111            Some(Frame::Chunks(major_type)) => Some(*major_type),
112            _ => None,
113        }
114    }
115
116    /// Account for a token, or fail without changing anything.
117    fn settle(&mut self, token: &Token) -> Result<(), Error> {
118        if let Some(expected) = self.chunk_type() {
119            let found = token_major_type(token);
120            let is_chunk = matches!(token, Token::Bytes(_) | Token::Text(_)) && found == expected;
121            if !is_chunk && !matches!(token, Token::Break) {
122                return Err(Error::InvalidChunkMajorType { expected, found });
123            }
124        }
125        match token {
126            Token::Break => {
127                let closes = matches!(
128                    self.frames.last(),
129                    Some(
130                        Frame::Array(None)
131                            | Frame::Chunks(_)
132                            | Frame::Map {
133                                remaining: None,
134                                value_next: false,
135                            }
136                    )
137                );
138                if !closes {
139                    return Err(Error::UnexpectedBreak);
140                }
141                self.frames.pop();
142                self.finish_item();
143            }
144            Token::ArrayStart(Some(0)) | Token::MapStart(Some(0)) => self.finish_item(),
145            Token::ArrayStart(count) => self.open(Frame::Array(*count))?,
146            Token::MapStart(count) => {
147                self.open(Frame::Map {
148                    remaining: *count,
149                    value_next: false,
150                })?;
151            }
152            Token::Tag(_) => self.open(Frame::Tag)?,
153            Token::BytesStart => self.open(Frame::Chunks(2))?,
154            Token::TextStart => self.open(Frame::Chunks(3))?,
155            // a chunk completes nothing on its own
156            Token::Bytes(_) | Token::Text(_) if self.chunk_type().is_some() => {}
157            _ => self.finish_item(),
158        }
159        Ok(())
160    }
161
162    fn open(&mut self, frame: Frame) -> Result<(), Error> {
163        // the top-level item takes one level, as in the decoder
164        if self.frames.len() + 1 >= self.limit {
165            return Err(Error::RecursionLimit { limit: self.limit });
166        }
167        self.frames.push(frame);
168        Ok(())
169    }
170
171    /// One item completed: close every frame it completes.
172    fn finish_item(&mut self) {
173        loop {
174            match self.frames.last_mut() {
175                Some(Frame::Tag) => {
176                    self.frames.pop();
177                }
178                Some(Frame::Array(Some(remaining))) => {
179                    *remaining -= 1;
180                    if *remaining > 0 {
181                        return;
182                    }
183                    self.frames.pop();
184                }
185                Some(Frame::Map {
186                    remaining,
187                    value_next,
188                }) => {
189                    *value_next = !*value_next;
190                    if *value_next {
191                        return;
192                    }
193                    match remaining {
194                        Some(0) => unreachable!("a map with no pair left is closed"),
195                        Some(count) => {
196                            *count -= 1;
197                            if *count > 0 {
198                                return;
199                            }
200                            self.frames.pop();
201                        }
202                        None => return,
203                    }
204                }
205                Some(Frame::Array(None) | Frame::Chunks(_)) | None => return,
206            }
207        }
208    }
209}

211/// The major type a token is written with.
212fn token_major_type(token: &Token) -> u8 {
213    match token {
214        Token::Unsigned(_) => 0,
215        Token::Signed(_) => 1,
216        Token::Bytes(_) | Token::BytesStart => 2,
217        Token::Text(_) | Token::TextStart => 3,
218        Token::ArrayStart(_) => 4,
219        Token::MapStart(_) => 5,
220        Token::Tag(_) => 6,
221        _ => 7,
222    }
223}

225/// Decodes bytes into [`Token`]s as they arrive, from [`Decoder::stream`].
226///
227/// After an error it yields nothing more.
228#[derive(Debug, Clone)]
229pub struct StreamDecoder {
230    buffer: Vec<u8>,
231    consumed: usize,
232    nesting: Nesting,
233    items: usize,
234    item_limit: usize,
235    allow_indefinite: bool,
236    failed: bool,
237}
238
239impl StreamDecoder {
240    pub(crate) fn new(nesting_limit: usize, item_limit: usize, allow_indefinite: bool) -> Self {
241        Self {
242            buffer: vec![],
243            consumed: 0,
244            nesting: Nesting::new(nesting_limit),
245            items: 0,
246            item_limit,
247            allow_indefinite,
248            failed: false,
249        }
250    }
251
252    /// Add bytes to decode.
253    pub fn feed(&mut self, bytes: &[u8]) {
254        // drop handed out bytes once they outweigh what is waiting
255        if self.consumed > 0 && self.consumed >= self.buffer.len() / 2 {
256            self.buffer.drain(..self.consumed);
257            self.consumed = 0;
258        }
259        self.buffer.extend_from_slice(bytes);
260    }
261
262    /// Bytes fed but not yet handed out as tokens.
263    #[must_use]
264    pub fn pending(&self) -> usize {
265        self.buffer.len() - self.consumed
266    }
267
268    /// Items open now.
269    #[must_use]
270    pub fn depth(&self) -> usize {
271        self.nesting.frames.len()
272    }
273
274    /// Whether every item started has ended and no byte is waiting, so a
275    /// sequence may end here.
276    #[must_use]
277    pub fn is_at_boundary(&self) -> bool {
278        !self.failed && self.nesting.frames.is_empty() && self.pending() == 0
279    }
280
281    /// The next token, or `None` until more bytes are fed.
282    ///
283    /// # Errors
284    /// If the input is not well formed or breaks a limit of the decoder
285    pub fn next_token(&mut self) -> Result<Option<Token>, Error> {
286        if self.failed {
287            return Ok(None);
288        }
289        let result = self.read_token().and_then(|read| {
290            let Some((token, used)) = read else {
291                return Ok(None);
292            };
293            self.nesting.settle(&token)?;
294            self.consumed += used;
295            // a break is not an item, as in the decoder
296            if token != Token::Break {
297                self.items += 1;
298            }
299            Ok(Some(token))
300        });
301        if result.is_err() {
302            self.failed = true;
303        }
304        result
305    }
306
307    /// The token at the front and its length, or `None` if incomplete.
308    fn read_token(&self) -> Result<Option<(Token, usize)>, Error> {
309        let bytes = &self.buffer[self.consumed..];
310        let Some(&head) = bytes.first() else {
311            return Ok(None);
312        };
313        if head == BREAK {
314            return Ok(Some((Token::Break, 1)));
315        }
316        if self.items >= self.item_limit {
317            return Err(Error::ItemLimit {
318                limit: self.item_limit,
319            });
320        }
321        let major_type = head >> 5;
322        let additional = head & 0b0001_1111;
323        // chunks are definite strings of the string's major type
324        if let Some(expected) = self.nesting.chunk_type() {
325            if major_type != expected {
326                return Err(Error::InvalidChunkMajorType {
327                    expected,
328                    found: major_type,
329                });
330            }
331            if additional == 31 {
332                return Err(Error::InvalidAdditionalInfo {
333                    major_type,
334                    additional,
335                });
336            }
337        }
338        if additional == 31 {
339            return self
340                .start_indefinite(major_type)
341                .map(|token| Some((token, 1)));
342        }
343        if major_type == 7 {
344            let read = read_simple_or_float(additional, &bytes[1..])?;
345            return Ok(read.map(|(token, used)| (token, used + 1)));
346        }
347        let Some((argument, used)) = read_argument(major_type, additional, &bytes[1..])? else {
348            return Ok(None);
349        };
350        let used = used + 1;
351        let token = match major_type {
352            0 => Token::Unsigned(argument),
353            1 => Token::Signed(argument),
354            2 | 3 => {
355                let Some(payload) = take(&bytes[used..], argument) else {
356                    return Ok(None);
357                };
358                let used = used + payload.len();
359                if major_type == 2 {
360                    return Ok(Some((Token::Bytes(payload.to_vec()), used)));
361                }
362                let text = String::from_utf8(payload.to_vec())?;
363                return Ok(Some((Token::Text(text), used)));
364            }
365            4 => Token::ArrayStart(Some(argument)),
366            5 => Token::MapStart(Some(argument)),
367            _ => Token::Tag(argument),
368        };
369        Ok(Some((token, used)))
370    }
371
372    /// The token for an indefinite length head.
373    fn start_indefinite(&self, major_type: u8) -> Result<Token, Error> {
374        if !matches!(major_type, 2..=5) {
375            return Err(Error::InvalidAdditionalInfo {
376                major_type,
377                additional: 31,
378            });
379        }
380        if !self.allow_indefinite {
381            return Err(Error::IndefiniteNotAllowed);
382        }
383        Ok(match major_type {
384            2 => Token::BytesStart,
385            3 => Token::TextStart,
386            4 => Token::ArrayStart(None),
387            _ => Token::MapStart(None),
388        })
389    }
390}

392/// A head's argument and its length, or `None` if incomplete.
393fn read_argument(
394    major_type: u8,
395    additional: u8,
396    bytes: &[u8],
397) -> Result<Option<(u64, usize)>, Error> {
398    let width = match additional {
399        0..=23 => return Ok(Some((u64::from(additional), 0))),
400        24 => 1,
401        25 => 2,
402        26 => 4,
403        27 => 8,
404        _ => {
405            return Err(Error::InvalidAdditionalInfo {
406                major_type,
407                additional,
408            });
409        }
410    };
411    let Some(bytes) = bytes.get(..width) else {
412        return Ok(None);
413    };
414    let argument = bytes
415        .iter()
416        .fold(0_u64, |argument, byte| argument << 8 | u64::from(*byte));
417    Ok(Some((argument, width)))
418}

420/// The first `length` bytes, if there are that many.
421fn take(bytes: &[u8], length: u64) -> Option<&[u8]> {
422    bytes.get(..usize::try_from(length).ok()?)
423}

425/// A major type 7 token and its length after the head, or `None` if
426/// incomplete.
427fn read_simple_or_float(additional: u8, bytes: &[u8]) -> Result<Option<(Token, usize)>, Error> {
428    let token = match additional {
429        0..=19 => Token::Simple(Simple::try_from(additional)?),
430        20 => Token::Boolean(false),
431        21 => Token::Boolean(true),
432        22 => Token::Null,
433        23 => Token::Undefined,
434        24 => {
435            let Some(&value) = bytes.first() else {
436                return Ok(None);
437            };
438            if value < 32 {
439                return Err(Error::InvalidSimpleValue(value));
440            }
441            return Ok(Some((Token::Simple(Simple::try_from(value)?), 1)));
442        }
443        25 => {
444            let Some(&[first, second]) = bytes.first_chunk::<2>() else {
445                return Ok(None);
446            };
447            let number = f16_to_f64(half::f16::from_be_bytes([first, second]));
448            return Ok(Some((Token::Floating(number), 2)));
449        }
450        26 => {
451            let Some(bits) = bytes.first_chunk::<4>() else {
452                return Ok(None);
453            };
454            let number = f32_to_f64(f32::from_be_bytes(*bits));
455            return Ok(Some((Token::Floating(number), 4)));
456        }
457        27 => {
458            let Some(bits) = bytes.first_chunk::<8>() else {
459                return Ok(None);
460            };
461            return Ok(Some((Token::Floating(f64::from_be_bytes(*bits)), 8)));
462        }
463        _ => {
464            return Err(Error::InvalidAdditionalInfo {
465                major_type: 7,
466                additional,
467            });
468        }
469    };
470    Ok(Some((token, 0)))
471}

473/// Writes [`Token`]s as bytes, refusing any that would not be well formed.
474///
475/// Numbers are written in their shortest form. Repeated map keys and tag
476/// content are not checked.
477///
478/// # Example
479/// ```rust
480/// use cbor_next::{StreamEncoder, Token};
481///
482/// let mut encoder = StreamEncoder::new();
483/// encoder.push(&Token::ArrayStart(None)).unwrap();
484/// encoder.push(&Token::Unsigned(1)).unwrap();
485/// // a break cannot stand where a map key is due
486/// encoder.push(&Token::MapStart(Some(1))).unwrap();
487/// assert!(encoder.push(&Token::Break).is_err());
488/// ```
489#[derive(Debug, Clone)]
490pub struct StreamEncoder {
491    buffer: Vec<u8>,
492    nesting: Nesting,
493}
494
495impl Default for StreamEncoder {
496    fn default() -> Self {
497        Self::new()
498    }
499}
500
501impl StreamEncoder {
502    /// Create an encoder with the decoder's default nesting limit.
503    #[must_use]
504    pub fn new() -> Self {
505        Self {
506            buffer: vec![],
507            nesting: Nesting::new(Decoder::DEFAULT_MAX_NESTING_DEPTH),
508        }
509    }
510
511    /// Set the nesting limit.
512    #[must_use]
513    pub fn max_nesting_depth(mut self, depth: usize) -> Self {
514        self.nesting.limit = depth;
515        self
516    }
517
518    /// Write a token; on error nothing is written.
519    ///
520    /// # Errors
521    /// If the token is out of place, such as a break with nothing open to
522    /// close, or opens past the nesting limit
523    pub fn push(&mut self, token: &Token) -> Result<(), Error> {
524        self.nesting.settle(token)?;
525        let buffer = &mut self.buffer;
526        match token {
527            Token::Unsigned(number) => write_head(buffer, 0, *number),
528            Token::Signed(number) => write_head(buffer, 1, *number),
529            Token::Bytes(bytes) => {
530                write_head(buffer, 2, count_of(bytes.len()));
531                buffer.extend_from_slice(bytes);
532            }
533            Token::Text(text) => {
534                write_head(buffer, 3, count_of(text.len()));
535                buffer.extend_from_slice(text.as_bytes());
536            }
537            Token::BytesStart => buffer.push(0x5f),
538            Token::TextStart => buffer.push(0x7f),
539            Token::ArrayStart(Some(count)) => write_head(buffer, 4, *count),
540            Token::ArrayStart(None) => buffer.push(0x9f),
541            Token::MapStart(Some(count)) => write_head(buffer, 5, *count),
542            Token::MapStart(None) => buffer.push(0xbf),
543            Token::Tag(number) => write_head(buffer, 6, *number),
544            Token::Boolean(false) => buffer.push(0xf4),
545            Token::Boolean(true) => buffer.push(0xf5),
546            Token::Null => buffer.push(0xf6),
547            Token::Undefined => buffer.push(0xf7),
548            Token::Floating(number) => write_float(buffer, 7, *number),
549            Token::Simple(simple) => {
550                let value = simple.value();
551                if value <= 23 {
552                    buffer.push(0xe0 | value);
553                } else {
554                    buffer.extend_from_slice(&[0xf8, value]);
555                }
556            }
557            Token::Break => buffer.push(BREAK),
558        }
559        Ok(())
560    }
561
562    /// The bytes written so far.
563    #[must_use]
564    pub fn bytes(&self) -> &[u8] {
565        &self.buffer
566    }
567
568    /// Take the bytes written so far, leaving the encoder empty.
569    pub fn take(&mut self) -> Vec<u8> {
570        std::mem::take(&mut self.buffer)
571    }
572
573    /// Items open now.
574    #[must_use]
575    pub fn depth(&self) -> usize {
576        self.nesting.frames.len()
577    }
578
579    /// Whether every item started has ended.
580    #[must_use]
581    pub fn is_at_boundary(&self) -> bool {
582        self.nesting.frames.is_empty()
583    }
